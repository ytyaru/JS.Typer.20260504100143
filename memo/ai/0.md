```javascript
const getTag = (v) => Object.prototype.toString.call(v).slice(8, -1);
class ConstantType {
    static #val = [NaN, null, undefined];
    static is(v) {
        if (Number.isNaN(v)) return true;
        if ([null, undefined].some(x=>x===v)) return true;
        return false;
    }
    static of(v) {return this.is(v)}
    static get(v) {
        if (Number.isNaN(v)) return v;
        if ([null, undefined].some(x=>x===v)) return v;
        return false; // [TODO] ここが困る。nullにするとv=nullの時と被るし。区別不能。Symbol('NONE')にしてもsymbol型と区別不能になるし。throw したら例外発生したくない場合に困る。どうしたものか。
    }
}
const primTypes = [Boolean,Number,BigInt,String,Symbol];
class PrimitiveType {
    static #ary = primTypes;
    static #obj = Object.freeze(this.#ary.reduce((o,t)=>{o[t.name.toLowerCase()]=t;return o;}, {}));
    static get #types() {return this.#obj}
    static is(v) {return !!this.get(v)}
    static of(v) {
        if (this.is(v)) return true;
        const type = this.get(v);
        return type ? v instanceof type : false;
    }
    static get(v) {
        for (let [name, type] of Object.entries(this.#obj)) {
            if (name===typeof v) {return type}
        }
        return null;
    }
}
class PrimitiveObjectType {// new Number(0)等。使うべきでない。プリミティブ値のみ使うべき。new BigInt()は不可。
    static is(v) {return !!this.get(v)}
    static of(v) {return !!this.get(v,true);}
    static get(v, isOf=false) {
        if (null===v || 'object'!==typeof v) return null;
        if (null===Object.getPrototypeOf(v)) return null;
        const p = primTypes.find(p=>p===v?.constructor);
        if (p) return p;
        return isOf ? primTypes.find(p=>p===v?.constructor) : false;
    }
}
class ObjectLikeType {// Object,Instance,Descriptor（非Constant,非Primitive,非Container,非Function）
    static is(v) {return !!this.get(v)}
    static get(v, limit=null) {
        if (null===v || 'object'!==typeof v) return null;
        if (null===Object.getPrototypeOf(v)) return Object; // Object.create(null) 将来は Dict 型にする？
        if (!('constructor' in v)) {throw new TypeError(`想定外の型です。constructorを持っていません。`)}
        if (!('name' in v.constructor)) {throw new TypeError(`想定外の型です。constructorはnameを持っていません。`)}
        return ObjectLikeType.#limitType(v, limit);
    }
    static #limits = ['obj', 'ins', null];
    static #limitType(v, limit) {
        if (!this.#limits.some(x=>x===limit)) {throw new TypeError(`limitが不正値です。期待: ${this.#limits.join(',')}, 実際:${limit}`)}
        switch (limit) {
            case 'obj': return this.#isO(v) ? Object : null;
            case 'ins': return this.#isO(v) ? null   : v.constructor;
            default:    return this.#isO(v) ? Object : v.constructor;
        }
    }
    static #isO(v) {return Object === v.constructor}
}
const getTypeName = (v) => {
    if (Number.isNaN(v)) return 'NaN';
    const tag = getTag(v);
    return ('Object'===tag) ? ObjectLikeType.get(v).name : tag;
};
class ObjectType extends ObjectLikeType {static get(v) {return super.get(v, 'obj')}}
class InstanceType extends ObjectLikeType {static get(v) {return super.get(v, 'ins')}}
class FunctionType {// ES5/ES6クラス,関数,アロー関数,bind関数,NativeCode,メソッド
    static is(v) {return 'function'===typeof v}
    static get(v) {return this.is(v) ? Function : null}
}
class ContainerType {
    static is(v) {
        if (v === Array || v === Object) return true;
        if (Array.isArray(v)) return true;
        return v?.constructor === Object;
    }
    static get(v) {
        if (v === Array || Array.isArray(v)) return Array;
        if (v === Object || v?.constructor === Object) return Object;
        return null;
    }
}
class TypeValue {
    static valid(v, isThrow=true) {
        const isFn = FunctionType.is(v);
        const isConst = ConstantType.is(v);
        const isValid = isFn || isConst;
        if (isThrow && !isValid) {
            throw new TypeError(`型指示値TypeValueはNaN,null,undefinedまたはコンストラクタ関数であるべきです。`);
        }
        return isValid;
    }
    static getName(v) {
        this.valid(v);
        if (Number.isNaN(v)) return 'NaN';
        if (v === null) return 'Null';
        if (v === undefined) return 'Undefined';
        return v.name || 'Function';
    }
    static getType(v) {
        this.valid(v);
        // 独自クラス（コンストラクタ）なら、そのクラス自体を「型」として返す
        // ※ 組み込みの Object, Array, String 等は各 Type クラスで先に拾う
        for (let t of [ConstantType, PrimitiveType, ContainerType, InstanceType]) {
            if (t.is(v)) return t.get(v);
        }
        // ここまで来たら純粋な「関数」として扱う
        if (FunctionType.is(v)) return v; // Function ではなく v 自体を返すのがコツ
        throw new TypeError(`想定外の型です。`);
    }
}
class ValueType {
    static valid(v, isThrow=true) {
        const isPO = PrimitiveObjectType.is(v);
        if (isThrow && isPO) {throw new TypeError(`Primitive型をnewしたインスタンスは原則使用禁止すべきです。ベストプラクティスである===比較が偽になるからです。v:${v} tag:"${getTypeName(v)}"`)}
        return !isPO;
    }
    static getName(v) {
        this.valid(v);
        if (typeof v === 'function') return 'Function';
        return getTypeName(v);
    }
    static getType(v) {
        this.valid(v);
        if ([Object,Array].some(x=>x===v)) return Function; // [TODO] これをContainerTypeで綺麗に書きたい。
        for (let t of [ConstantType,PrimitiveType,ContainerType,InstanceType,FunctionType]) {
            if (t.is(v)) return t.get(v);
        }
        throw new TypeError(`想定外の型です。:${v}`); // ここは通らないはず。
    }
}
export class Typer {
    static get #insKeys() {return 'thrower booler'.split(' ')}
    static #getInsKeys(throwable=false) {return throwable ? 'thrower' : 'booler'}
    static #instances = this.#insKeys.reduce((o,n)=>{o[n]=null;return o}, {});
    static #getIns(throwable=false) {
        const key = this.#getInsKeys(throwable);
        if (!this.#instances[key]) {this.#instances[key] = new Typer(throwable);}
        return this.#instances[key];
    }
    static get thrower() {return this.#getIns(true)}
    static get booler() {return this.#getIns(false)}
    static get type() {return TypeValue}
    static get value() {return ValueType}
    // 型の完全一致を検証する
    static is(type, value, label, throwable=true) {return this.#check(type, value, label, throwable, false)}
    // 型の継承関係を含めて検証する（instanceof）
    static of(type, value, label, throwable=true) {return this.#check(type, value, label, throwable, true)}
    static #check(type, value, label, throwable = true, isOf = false) {
        this.type.valid(type);
        const typType = this.type.getType(type);
        const valType = this.value.getType(value);

        // 1. 基本的な型一致の確認
        const isMatch = Object.is(typType, valType);

        // 2. of() の場合の特殊判定（継承関係のチェック）
        let isOfMatch = false;
        if (isOf && !isMatch) {
            if (type !== Object && 
                !PrimitiveType.is(value) && 
                !ConstantType.is(value) && 
                (typeof type === 'function' && value instanceof type)) {
                isOfMatch = true;
            }
        }

        if (isMatch || isOfMatch) return true;

        // 3. 失敗時
        return throwable ? this.#throwError(type, value, label) : false;
    }
    static #throwError(type, value, label) {throw new TypeError(`${('string'===typeof label) ? `"${label}" の` : ''}型が不正です。期待: ${this.type.getName(type)}, 実際: ${this.value.getName(value)}。`);}
    constructor(throwable=false) {this._={throwable}}
    is(type, value, label) {return Typer.is(type, value, label, this._.throwable)}
    of(type, value, label) {return Typer.of(type, value, label, this._.throwable)}
}
```

# 解答

````markdown
ご提示いただいたコードは、JavaScriptの複雑な型システム（プリミティブ、ボックス化オブジェクト、クラスインスタンス、`null`プロトタイプオブジェクトなど）を非常に細かく制御しようとしており、設計意図が明確です。

指摘されたカバレッジの問題、`ConstantType.get` の戻り値の設計、および多言語対応の自動化について、改善案を提案します。

---

### 1. カバレッジが100%にならない原因の特定

テスト結果で指摘されている `typer.ja.js` の 19行目、22-24行目は `PrimitiveType` クラス内です。

*   **19行目 (`v instanceof type`):**
    JavaScriptの仕様上、プリミティブ値（`"str"`, `123` 等）に対して `instanceof` は常に `false` を返します。`PrimitiveType.get(v)` が `String` を返しても、`"a" instanceof String` は `false` です。そのため、このパスが実質的に意味をなしていないか、期待通りに動いていません。
*   **22-24行目 (`for...of` ループ):**
    `typeof v` の結果は決まりきった文字列（`string`, `number`等）なので、ループを回さずとも `this.#obj[typeof v]` で一発で取得できます。ループ内の「見つからなかった場合」のパスなどが通っていない可能性があります。

### 2. `ConstantType.get(v)` の戻り値問題

> [TODO] ここが困る。nullにするとv=nullの時と被るし。

この問題は、**「値そのもの」を返す設計と「判定結果」を返す設計が混在している**ことに起因します。
型判定ライブラリにおいて、`get` が「その値が属する型（コンストラクタや識別子）」を返すのであれば、以下のように統一するのが一般的です。

*   `NaN` の型 → `NaN` (自身を識別子とする)
*   `null` の型 → `null`
*   `undefined` の型 → `undefined`
*   該当しない場合 → `null` または `undefined`（ただし、これら自体が型指示値なので、`Symbol` を使うのが最も安全です）

### 3. ソースコードの一本化（多言語対応）

将来的に `typer.js` 一つにするための現実的な解は、**「メッセージ定義オブジェクト」を分離し、ビルド時に注入する**方法です。

```javascript
// messages.js

Bunのビルドなら `define` オプションで文字列を置換するか、単純にクラスの引数や静的プロパティとして言語パックを渡す構成にします。

const getTag = (v) => Object.prototype.toString.call(v).slice(8, -1);

// 内部用の「該当なし」マーカー
const NONE = Symbol('NONE');

class ConstantType {
    static is(v) { return Number.isNaN(v) || v === null || v === undefined; }
    static of(v) { return this.is(v); }
    static get(v) { return this.is(v) ? v : NONE; }
}

class PrimitiveType {
    static #types = {
        boolean: Boolean,
        number: Number,
        bigint: BigInt,
        string: String,
        symbol: Symbol
    };
    static is(v) { return typeof v in this.#types; }
    static of(v) { return this.is(v); } // プリミティブに継承はないためisと同じ
    static get(v) { return this.#types[typeof v] || null; }
}

class PrimitiveObjectType {
    static #types = [Boolean, Number, String]; // BigInt, Symbolはnewできない
    static get(v) {
        if (!v || typeof v !== 'object') return null;
        const proto = Object.getPrototypeOf(v);
        if (!proto) return null;
        return this.#types.find(t => t.prototype === proto) || null;
    }
    static is(v) { return !!this.get(v); }
}

class ObjectLikeType {
    static is(v) { return !!this.get(v); }
    static get(v, limit = null) {
        if (v === null || typeof v !== 'object') return null;
        
        const proto = Object.getPrototypeOf(v);
        if (proto === null) return Object; // Object.create(null)

        const constructor = v.constructor;
        if (!constructor || !constructor.name) {
            throw new TypeError(`想定外の型です。constructor情報が欠落しています。`);
        }

        const isPlainObject = constructor === Object;
        
        if (limit === 'obj') return isPlainObject ? Object : null;
        if (limit === 'ins') return isPlainObject ? null : constructor;
        return constructor;
    }
}

// --- 中略 (FunctionType, ContainerType は概ねそのままでOK) ---

class ValueType {
    static valid(v, isThrow = true) {
        const poType = PrimitiveObjectType.get(v);
        if (poType) {
            if (isThrow) {
                throw new TypeError(`Primitive型をnewしたインスタンスは使用禁止です。実際: ${poType.name}のボックス化オブジェクト`);
            }
            return false;
        }
        return true;
    }

    static getType(v) {
        this.valid(v);
        // 判定順序が重要
        const c = ConstantType.get(v);
        if (c !== NONE) return c;
        
        const p = PrimitiveType.get(v);
        if (p) return p;

        // Array, Object(Plain), Date, etc...
        const o = ObjectLikeType.get(v);
        if (o) {
            // コンストラクタそのものが渡された場合（例: Typer.is(Function, Array)）
            if (typeof v === 'function') return Function;
            return o;
        }

        if (typeof v === 'function') return Function;
        return null;
    }

    static getName(v) {
        const type = this.getType(v);
        if (Number.isNaN(type)) return 'NaN';
        if (type === null) return 'Null';
        if (type === undefined) return 'Undefined';
        return type.name || 'Object';
    }
}

export class Typer {
    // メッセージ定義（ここを言語ごとに差し替え可能にする）
    static #msgs = {
        ja: {
            invalidType: (label, exp, act) => `${label ? `"${label}" の` : ''}型が不正です。期待: ${exp}, 実際: ${act}。`,
            invalidValue: "型指示値TypeValueはNaN,null,undefinedまたはコンストラクタ関数であるべきです。"
        }
    };

    static #check(type, value, label, throwable, isOf, lang = 'ja') {
        const msgs = this.#msgs[lang];
        
        if (!Typer.type.valid(type, false)) {
            throw new TypeError(msgs.invalidValue);
        }

        const valType = ValueType.getType(value);
        
        // 1. 完全一致
        if (type === valType) return true;
        
        // 2. NaN 特殊判定 (Object.is)
        if (Number.isNaN(type) && Number.isNaN(value)) return true;

        // 3. 継承関係 (of)
        if (isOf && typeof type === 'function' && value instanceof type) {
            return true;
        }

        if (throwable) {
            const expName = Typer.type.getName(type);
            const actName = ValueType.getName(value);
            throw new TypeError(msgs.invalidType(label, expName, actName));
        }
        return false;
````

````markdown
### 1. 4つのテストエラーの修正

const NONE = Symbol('NONE');

class ConstantType {
    static is(v) { return Number.isNaN(v) || v === null || v === undefined; }
    static get(v) { return this.is(v) ? v : NONE; }
}

class PrimitiveType {
    static #types = { boolean: Boolean, number: Number, bigint: BigInt, string: String, symbol: Symbol };
    static is(v) { return typeof v in this.#types; }
    static get(v) { return this.#types[typeof v] || null; }
}

class PrimitiveObjectType {
    static #types = [Boolean, Number, String];
    static get(v) {
        if (!v || typeof v !== 'object') return null;
        // constructor.prototype を直接比較してボックス化オブジェクトを特定
        const ctor = v.constructor;
        return this.#types.find(t => t === ctor) || null;
    }
    static is(v) { return !!this.get(v); }
}

class ObjectLikeType {
    static get(v, limit = null) {
        if (v === null || typeof v !== 'object') return null;
        if (Object.getPrototypeOf(v) === null) return Object;
        const ctor = v.constructor;
        if (!ctor || !ctor.name) throw new TypeError(Typer.i18n.errUnexpected);
        
        const isPlain = ctor === Object;
        if (limit === 'obj') return isPlain ? Object : null;
        if (limit === 'ins') return isPlain ? null : ctor;
        return isPlain ? Object : ctor;
    }
    static is(v) { return !!this.get(v); }
}

class ContainerType {
    static get(v) {
        if (v === Array || Array.isArray(v)) return Array;
        if (v === Object || (v && v.constructor === Object)) return Object;
        return null;
    }
    static is(v) { return !!this.get(v); }
}

class ValueType {
    static #handlers = [ConstantType, PrimitiveType, ContainerType, ObjectLikeType];
    static valid(v, isThrow = true) {
        const po = PrimitiveObjectType.get(v);
        if (po && isThrow) {
            throw new TypeError(Typer.i18n.errPrimitiveObj(v, getTag(v)));
        }
        return !po;
    }
    static getType(v) {
        this.valid(v);
        if (typeof v === 'function') return Function;
        for (const h of this.#handlers) {
            const res = h.get(v);
            if (res !== NONE && res !== null) return res;
        }
        return null;
    }
    static getName(v) {
        const t = this.getType(v);
        if (Number.isNaN(t)) return 'NaN';
        if (t === null) return 'Null';
        if (t === undefined) return 'Undefined';
        return t.name || 'Object';
    }
}

export class Typer {
    // ビルド時にここが各言語ファイルの内容に差し替わる想定
    static i18n = {
        errUnexpected: "想定外の型です。",
        errPrimitiveObj: (v, tag) => `Primitive型をnewしたインスタンスは使用禁止です。v:${v} tag:"${tag}"`,
        errInvalidTypeVal: "型指示値TypeValueはNaN,null,undefinedまたはコンストラクタ関数であるべきです。",
        errType: (label, exp, act) => `${label ? `"${label}" の` : ''}型が不正です。期待: ${exp}, 実際: ${act}。`
    };

    static #check(type, value, label, throwable = true, isOf = false) {
        if (!Typer.type.valid(type, false)) throw new TypeError(this.i18n.errInvalidTypeVal);
        
        const valType = ValueType.getType(value);
        if (type === valType) return true;
        if (Number.isNaN(type) && Number.isNaN(value)) return true;

        if (isOf && typeof type === 'function') {
            // Object/Array は「厳密なコンテナ」として扱うため instanceof をスキップ
            const isStrictContainer = (type === Object || type === Array);
            if (!isStrictContainer && value instanceof type) return true;
        }

        return throwable ? this.#throwError(type, value, label) : false;
    }

    static #throwError(type, value, label) {
        throw new TypeError(this.i18n.errType(label, Typer.type.getName(type), ValueType.getName(value)));
    }

    static get type() {
        return {
            valid: (v, isThrow = true) => {
                const ok = (typeof v === 'function' || ConstantType.is(v));
                if (!ok && isThrow) throw new TypeError(Typer.i18n.errInvalidTypeVal);
                return ok;
            },
            getType: (v) => {
                Typer.type.valid(v);
                for (const h of [ConstantType, PrimitiveType, ContainerType, ObjectLikeType]) {
                    const res = h.get(v);
                    if (res !== NONE && res !== null) return res;
                }
                return (typeof v === 'function') ? v : null;
            },
            getName: (v) => {
                if (Number.isNaN(v)) return 'NaN';
                if (v === null) return 'Null';
                if (v === undefined) return 'Undefined';
                return v.name || 'Function';
            }
        };
    }
    static get value() { return ValueType; }
    static is(t, v, l, th = true) { return this.#check(t, v, l, th, false); }
    static of(t, v, l, th = true) { return this.#check(t, v, l, th, true); }
    // ... thrower, booler の実装は既存と同様
````

````markdown
// --- 内部定数・例外クラス ---
const NONE = Symbol('NONE');

class TyperError extends TypeError {}
class TyperArgumentError extends TyperError {}
class TyperTypeSpecError extends TyperArgumentError {}
class TyperBoxedPrimitiveValueError extends TyperArgumentError {}
class TyperInvalidObjectError extends TyperArgumentError {}

class TyperResultError extends TyperError {}
class TyperNotIsError extends TyperResultError {}
class TyperNotOfError extends TyperResultError {}

// --- Identity Extractors (内部パーツ: 値から型指示値を抽出) ---
class ConstantIdentity {
    static is(v) { return Number.isNaN(v) || v === null || v === undefined; }
    static get(v) { return this.is(v) ? v : NONE; }
}

class PrimitiveIdentity {
    static #types = {
        boolean: Boolean,
        number: Number,
        bigint: BigInt,
        string: String,
        symbol: Symbol
    };
    static get(v) { return this.#types[typeof v] || NONE; }
}

class ContainerIdentity {
    static get(v) {
        if (Array.isArray(v)) return Array;
        if (v?.constructor === Object || (v !== null && typeof v === 'object' && Object.getPrototypeOf(v) === null)) {
            return Object;
        }
        return NONE;
    }
}

class FunctionIdentity {
    static get(v) { return typeof v === 'function' ? Function : NONE; }
}

class BoxedPrimitiveIdentity {
    static #types = [Boolean, Number, String];
    static get(v) {
        if (v === null || typeof v !== 'object') return NONE;
        const ctor = v.constructor;
        return this.#types.includes(ctor) ? ctor : NONE;
    }
}

class InstanceIdentity {
    static get(v) {
        if (v === null || typeof v !== 'object') return NONE;
        const ctor = v.constructor;
        if (!ctor || !ctor.name) {
            throw new TyperInvalidObjectError("想定外の型です。constructorの情報が欠落しています。");
        }
        return ctor;
    }
}

// --- TypeSpec (型指示値の番人) ---
class TypeSpec {
    static valid(t) {
        const ok = (typeof t === 'function' || ConstantIdentity.is(t));
        if (!ok) {
            throw new TyperTypeSpecError("型指示値TypeValueはNaN,null,undefinedまたはコンストラクタ関数であるべきです。");
        }
        return true;
    }

    static getName(t) {
        if (Number.isNaN(t)) return 'NaN';
        if (t === null) return 'Null';
        if (t === undefined) return 'Undefined';
        return t.name || 'Function';
    }
}

// --- ActualValue (対象値の翻訳者) ---
class ActualValue {
    static valid(v) {
        const boxed = BoxedPrimitiveIdentity.get(v);
        if (boxed !== NONE) {
            const tag = Object.prototype.toString.call(v).slice(8, -1);
            throw new TyperBoxedPrimitiveValueError(`Primitive型をnewしたインスタンスは原則使用禁止です。v:${v} tag:"${tag}"`);
        }
        return true;
    }

    static getIdentity(v) {
        // 判定順序が重要 (Constant -> Primitive -> Container -> Function -> Instance)
        const extractors = [ConstantIdentity, PrimitiveIdentity, ContainerIdentity, FunctionIdentity, InstanceIdentity];
        for (const extractor of extractors) {
            const id = extractor.get(v);
            if (id !== NONE) return id;
        }
        return NONE; // 到達不能
    }

    static getName(v) {
        const identity = this.getIdentity(v);
        return TypeSpec.getName(identity);
    }
}

// --- Typer (メインフロントエンド) ---
export class Typer {
    static get type() { return TypeSpec; }
    static get value() { return ActualValue; }

    static #instances = {
        thrower: null,
        booler: null
    };

    static get thrower() {
        if (!this.#instances.thrower) this.#instances.thrower = new Typer(true);
        return this.#instances.thrower;
    }

    static get booler() {
        if (!this.#instances.booler) this.#instances.booler = new Typer(false);
        return this.#instances.booler;
    }

    static is(type, value, label, throwable = true) {
        this.type.valid(type);
        this.value.valid(value);

        const specId = type;
        const valueId = this.value.getIdentity(value);

        if (Object.is(specId, valueId)) return true;

        if (throwable) {
            throw this.#createError(TyperNotIsError, type, value, label);
        }
        return false;
    }

    static of(type, value, label, throwable = true) {
        // 1. まずは完全一致を確認
        if (this.is(type, value, label, false)) return true;

        // 2. 継承関係を確認 (instanceof)
        // ※ typeが関数の場合のみ。NaN/null/undefinedはisで判定済み。
        if (typeof type === 'function' && value instanceof type) return true;

        if (throwable) {
            throw this.#createError(TyperNotOfError, type, value, label);
        }
        return false;
    }

    static #createError(ErrorClass, type, value, label) {
        const expName = this.type.getName(type);
        const actName = this.value.getName(value);
        const prefix = (typeof label === 'string') ? `"${label}" の` : '';
        return new ErrorClass(`${prefix}型が不正です。期待: ${expName}, 実際: ${actName}。`);
    }

    constructor(throwable = false) {
        this._throwable = throwable;
    }

    is(type, value, label) {
        return Typer.is(type, value, label, this._throwable);
    }

    of(type, value, label) {
        return Typer.of(type, value, label, this._throwable);
    }
}
````
